function beginStickerDrag(template,sourceEl,e,existingId=null){
  e.preventDefault();e.stopPropagation();
  stickerDrag={template,sourceEl,existingId};sourceEl.classList.add('dragging');
  dragGhost.querySelector('strong').textContent=`${template.emoji} ${template.title}`;
  dragGhost.querySelector('small').textContent='零时长贴纸 · 5 分钟吸附';
  dragGhost.classList.add('show');moveGhost(e.clientX,e.clientY)
}
function previewStickerDrag(e){
  if(!stickerDrag)return;dragDidMove=true;
  const rect=planTrack.getBoundingClientRect();
  if(e.clientY<rect.top||e.clientY>rect.bottom){stickerDropPreview.className='sticker-drop-preview';return}
  const minute=snap(Math.max(DAY_START,Math.min(DAY_START+(e.clientY-rect.top+timelineWindow.scrollTop)/PX,DAY_END)));
  stickerDropPreview.style.top=`${(minute-DAY_START)*PX}px`;
  stickerDropPreview.querySelector('b').textContent=`${stickerDrag.template.emoji} ${fmt(minute)} · 不占时间`;
  stickerDropPreview.classList.add('show');
  stickerDrag.minute=minute
}
function finishStickerDrag(){
  if(!stickerDrag)return;
  const d=stickerDrag;d.sourceEl?.classList.remove('dragging');dragGhost.classList.remove('show');stickerDropPreview.className='sticker-drop-preview';
  if(Number.isFinite(d.minute)){
    if(d.existingId){
      const st=stickers.find(x=>x.id===d.existingId);if(st)st.anchorMinute=d.minute;
      toastMsg(`贴纸已移到 ${fmt(d.minute)}`)
    }else{
      stickers.push({id:`st-man-${++idSeq}`,templateId:d.template.id,title:d.template.title,emoji:d.template.emoji,color:d.template.color,anchorMinute:d.minute,status:'pending',origin:'manual',placementMode:'timeline'});
      toastMsg(`已放贴纸 · ${fmt(d.minute)}`)
    }
    renderStickers();setSaved()
  }
  stickerDrag=null
}
