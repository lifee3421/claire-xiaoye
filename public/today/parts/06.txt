function intervalsOverlap(a,b){return a.start<b.end&&a.end>b.start}
function planMove(segId,targetStart,allowRipple=true,allowCompression=true){
  const pair=segById(segId);if(!pair)return {type:'noop'};
  const duration=dur(pair.s),activeBlock=allBlocks().find(b=>b.id===segId);
  const others=allBlocks().filter(b=>b.id!==segId);
  const hard=others.filter(hardBlock).sort((a,b)=>a.start-b.start);
  const movable=others.filter(b=>b.kind==='task'&&!hardBlock(b)).sort((a,b)=>a.start-b.start);
  const start=Math.max(DAY_START,Math.min(snap(targetStart),DAY_END-duration)),range={start,end:start+duration};
  if(activeBlock&&activeBlock.start===start)return {type:'noop'};
  const hardHit=hard.find(b=>intervalsOverlap(range,b));
  if(hardHit){
    if(allowCompression&&hardHit.start>start)return {type:'needs-compression',boundary:hardHit,start,end:range.end,gapEnd:hardHit.start,requestedWork:pair.s.work,requestedRest:pair.s.rest};
    return {type:'hard-conflict',boundary:hardHit,start,end:range.end}
  }
  if(!allowRipple){
    const m=movable.find(b=>intervalsOverlap(range,b));
    if(m)return {type:'needs-compression',boundary:m,start,end:range.end,gapEnd:m.start,requestedWork:pair.s.work,requestedRest:pair.s.rest};
    return {type:'success-exact',positions:[{id:segId,start,end:range.end}],shifted:[]}
  }
  let cursor=range.end,shifted=[];
  for(const b of movable){
    if(b.end<=start||b.start>=cursor)continue;
    const next={id:b.id,start:cursor,end:cursor+b.duration};
    const hit=hard.find(h=>intervalsOverlap(next,h));
    if(hit||next.end>DAY_END)return {type:'hard-conflict',boundary:hit||{title:'上床边界',start:DAY_END,end:DAY_END},start,end:range.end};
    shifted.push(next);cursor=next.end
  }
  return {type:shifted.length?'success-ripple':'success-exact',positions:[{id:segId,start,end:range.end},...shifted],shifted}
}
