    if(s.placement!=='pool')return;
    const st=findNearestFree(s.id,NOW);if(st!=null){s.placement='timeline';s.start=st;delete s.lastTimelineStart;placed++}
  });
  renderAll();toastMsg(`已安排 ${placed} 块`);setSaved()
}
function clearFuture(){
  let n=0;groups.forEach(g=>g.segments.forEach(s=>{if(s.placement==='timeline'&&s.start>=NOW&&!locks.has(s.id)){s.lastTimelineStart=s.start;s.placement='pool';delete s.start;n++}}));renderAll();root.classList.add('arrange');toastMsg(`已清空未来 ${n} 块，全部回任务池`);setSaved()
}
