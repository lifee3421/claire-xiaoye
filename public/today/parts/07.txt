function findNearestFree(segId,preferred){
  const pair=segById(segId);if(!pair)return null;const d=dur(pair.s);
  const candidates=[];
  for(let s=DAY_START;s+d<=DAY_END;s+=5){
    const range={start:s,end:s+d};if(allBlocks().filter(b=>b.id!==segId).every(b=>!intervalsOverlap(range,b)))candidates.push(s)
  }
  return candidates.sort((a,b)=>Math.abs(a-preferred)-Math.abs(b-preferred))[0]??null
}

  const occupant=allBlocks().find(b=>preferred>=b.start&&preferred<b.end);
