function blockHTML(b){
  const current=b.start<=NOW&&b.end>NOW;
  const short=b.duration<=35;
  const medium=b.duration<70;
  const long=b.duration>=70;
  const lock=b.protected||b.locked;

  const rhythm=b.kind==='task'?`${b.work}${b.rest?`+${b.rest}`:''}`:`${b.duration}min`;
  const detail=b.kind==='task'
    ? (long?`${rhythm} · ${b.index}/${b.total} · P${b.priority}`:`${rhythm} · P${b.priority}`)
    : (meal?'生活卡':'保护节点');
  const tail=b.status==='completed'&&!meal?'✓':(!meal&&lock?'🔒':'');
  return `<div class="block ${classFor(b.cat)} ${current?'current':''} ${b.status==='completed'?'completed':''} ${lock?'locked':''} ${b.protected?'protected':''} ${short?'short':''} ${medium?'medium':''} ${long?'long':''}" data-block-id="${b.id}" style="top:${(b.start-DAY_START)*PX}px;height:${Math.max(20,b.duration*PX)}px">

    <button class="block-copy" type="button">
      <span class="block-time">${fmt(b.start)} – ${fmt(b.end)}</span>
      <strong>${esc(b.title)}</strong>
      <span class="block-meta"><span class="rhythm">${esc(detail)}</span>${!meal&&lock?'<span>🔒</span>':''}${b.status==='completed'?'<span>✓ 已完成</span>':''}</span>
    </button>
    ${meal?`<button class="meal-check ${b.status==='completed'?'done':''}" type="button" data-meal-toggle="${b.id}" aria-label="${b.status==='completed'?'恢复':'完成'}${esc(b.title)}">${b.status==='completed'?'✓':'○'}</button>`:`<span class="block-tail">${tail}</span>`}

  </div>`;
}

function computeTimelineGaps(){
  const occupied=allBlocks().filter(b=>b.status!=='cancelled').sort((a,b)=>a.start-b.start);
  const gaps=[];let cursor=DAY_START;
  for(const b of occupied){
    if(b.start>cursor&&b.start-cursor>=30)gaps.push({start:cursor,end:b.start,duration:b.start-cursor});
    cursor=Math.max(cursor,b.end)
  }
  if(DAY_END-cursor>=30)gaps.push({start:cursor,end:DAY_END,duration:DAY_END-cursor});
  return gaps;
}
