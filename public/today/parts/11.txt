function applyTemplate(id){
  const t=dayTemplates.find(x=>x.id===id);if(!t)return;
  currentTemplateId=t.id;closeSheet();toastMsg(`已应用「${t.name}」· Preview`);setSaved('模板已应用')
}
function saveTemplate(updateId){
  const name=(document.getElementById('templateName')?.value||'').trim();if(!name){toastMsg('先写模板名称');return}
  if(updateId&&updateId!=='new'){
    const t=dayTemplates.find(x=>x.id===updateId);if(t){t.name=name;t.description='由今天更新 · 独立模板草稿';toastMsg('当前模板已更新')}
  }else{
    const id=`tpl-custom-${++idSeq}`;dayTemplates.push({id,name,description:'从今天保存',scene:'school',wake:'07:30',bed:'23:20',default:false,builtin:false,content:{fixed:['午睡','午间启动缓冲','晚餐'],pool:['数学｜下一块','专业课｜下一块'],timeline:['当前时间线结构']}});currentTemplateId=id;toastMsg('已保存为新模板')
  }
  openTemplateHub();setSaved('模板已保存')
}

function saveNewTask(){
  const title=document.getElementById('newTitle').value.trim();if(!title){toastMsg('先写任务名称');return}
  const w=Math.max(5,Number(document.getElementById('newWork').value)||50),r=Math.max(0,Number(document.getElementById('newRest').value)||0),count=Math.max(1,Number(document.getElementById('newCount').value)||1),priority=Number(document.getElementById('newPriority').value.replace('P',''))||2,cat=document.getElementById('newCat').value,preferred=document.getElementById('newPreferred').value;
  groups.push({id:`newg-${++idSeq}`,title,cat,priority,splittable:count>1,preferred,segments:Array.from({length:count},()=>({id:`news-${++idSeq}`,work:w,rest:r,placement:'pool',status:'pending'}))});
  renderAll();closeSheet();root.classList.add('arrange');toastMsg('已加入今天任务池');setSaved()
}
